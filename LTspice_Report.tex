\section{S1 -- Reading the Model Card}
\circuit{s1-circuit.png}
  {LTspice testbench for the \texttt{.op} runs (\textit{L} = 65 nm and 70 nm).}
  {.step param L list 65nm 70nm\\
   .op\\
   .meas OP current FIND Id(M1) AT 65nm\\
   .meas OP current2 FIND Id(M1) AT 70nm}

\subsection{Extracted Card Parameters}
\begin{table}[H]
  \centering
  \begin{tabular}{@{}lc@{}}
    \toprule
    Parameter & Value \\
    \midrule
    XL (nm)   & -30  \\
    LINT (nm) & 5.25 \\
    WINT (nm) & 5    \\
    TOXE (nm) & 1.85 \\
    \bottomrule
  \end{tabular}
  \caption{Values read from the model card.}
\end{table}

\subsection{Calculations}
\begin{align*}
  L_{eff} &= L_{drawn} + XL - 2\,LINT = L_{drawn} - 40.5\ \text{nm}\\
  W_{eff} &= W_{drawn} - 2\,WINT = W_{drawn} - 10\ \text{nm}\\
  C_{ox}  &= \frac{3.9\,\varepsilon_0}{TOXE} = 1.87\ \mu\text{F/cm}^2
\end{align*}

\begin{table}[H]
  \centering
  \begin{tabular}{@{}lc@{}}
    \toprule
    $L$ (nm) & $L_{eff}$ (nm) \\
    \midrule
    65 & 24.5 \\
    70 & 29.5 \\
    \bottomrule
  \end{tabular}
  \caption{Computed effective channel length.}
\end{table}

\subsection{.op Results at $V_{GS}=V_{DS}=1.1$~V}
\begin{table}[H]
  \centering
  \begin{tabular}{@{}lc@{}}
    \toprule
    $L$ (nm) & $I_D$ (mA) \\
    \midrule
    65 & 1.15 \\
    70 & 1.05 \\
    \bottomrule
  \end{tabular}
  \caption{Drain current from \texttt{.op}.}
\end{table}

\begin{observations}
  We can see that with increase in channel length a proportional decrease in drain current.
  \[ \frac{L_{70}}{L_{65}} = 1.08 \]
  \[ \frac{I_{D65}}{I_{D70}} = \frac{1.15}{1.05} = 1.09 \]
  \[ \frac{L_{70}}{L_{65}} \approx \frac{I_{D65}}{I_{D70}} \]
\end{observations}

\begin{conclusion}
  From the observation we can conclude that \textbf{$I_D$ varies inversly proportional to channel length} which matches to the theory we learnt.
  \[ I_D \propto \frac{1}{L} \]
\end{conclusion}
